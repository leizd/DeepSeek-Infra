// This fixture initializes signed empty ownership, optionally configures an
// isolated backup policy, and validates settled output after deepseekd exits.
// It never seeds Agent input, schedules an Agent, or executes a node.
func TestNativeAgentDaemonPublicProcess(t *testing.T) {
	directory := os.Getenv("DEEPSEEK_AGENT_SIGNING_PROBE_DIR")
	stage := os.Getenv("DEEPSEEK_TEST_AGENT_DAEMON_STAGE")
	if directory == "" || (stage != "seed" && stage != "inspect") {
		t.Fatal("explicit isolated daemon stage required")
	}
	for _, name := range []string{"DEEPSEEK_WORKER_CONTROL_SIGNER_BUNDLE_FILE", "DEEPSEEK_WORKER_CONTROL_SIGNER_PASSPHRASE_FILE", "DEEPSEEK_API_KEY", "DEEPSEEK_WORKER_AUTHORITY_NOW"} {
		if _, exists := os.LookupEnv(name); exists {
			t.Fatal("private Rust configuration entered offline Go fixture", name)
		}
	}
	controlPath := filepath.Join(directory, "go-control")
	if stage == "seed" {
		control := nativeSigningControl(t, 30, controlPath)
		domains := []string{"agent_run", "action"}
		if os.Getenv("DEEPSEEK_TEST_AGENT_DAEMON_BROWSER") == "true" {
			nativeBrowserEmptyBackupOwnership(t, control)
			domains = append(domains, "policy", "target")
		}
		for _, domain := range domains {
			rows, err := control.ListAuthoritativeRecords(domain)
			if err != nil || len(rows) != 0 {
				t.Fatal("ownership fixture seeded application input", domain, err)
			}
		}
		browser := os.Getenv("DEEPSEEK_TEST_AGENT_DAEMON_BROWSER") == "true"
		if browser {
			nativeBrowserConfigureBackupPolicy(t, control)
		}
		if err := control.Close(); err != nil {
			t.Fatal(err)
		}
		nativeSigningWrite(t, filepath.Join(directory, "daemon-seed.json"), map[string]any{
			"controlPath": controlPath, "internalBearer": testInternalBearer,
			"promotionPublicKey": promotionRoutePublic, "fleetId": mutationFleetID,
			"environment": mutationEnvironment, "emptyApplicationRecords": !browser,
			"emptyAgentAndActionRecords": true, "backupPolicyConfiguredByGoFixture": browser})
		return
	}
	var input struct {
		RunID    string `json:"runId"`
		Browser  bool   `json:"browser"`
		Recovery bool   `json:"recovery"`
	}
	nativeSigningRead(t, filepath.Join(directory, "daemon-inspect-input.json"), &input)
	control, err := store.OpenControl(store.OpenOptions{Path: controlPath, Owner: "offline-daemon-output-validation",
		PromotionSignerPublicKey:    promotionRoutePublic,
		AgentOutcomeSignerPublicKey: os.Getenv("DEEPSEEK_WORKER_AUTHORITY_SIGNER_PUBLIC_KEY"),
		FleetID:                     mutationFleetID, Environment: mutationEnvironment})
	if err != nil {
		t.Fatal(err)
	}
	defer control.Close()
	metadata, found, err := control.AgentRunMetadata(input.RunID)
	expectedIndex, coderTask := int64(9), "write"
	if input.Browser {
		expectedIndex, coderTask = 10, "write through browser"
	}
	if input.Recovery {
		if !input.Browser {
			t.Fatal("browser recovery requires the actual browser scenario")
		}
		expectedIndex = 17
	}
	if err != nil || !found || metadata.Status != "done" || metadata.NextIndex != expectedIndex || !metadata.ConfirmPlan || metadata.Preset != "auto" {
		t.Fatal("actual daemon final metadata is incomplete", err)
	}
	expected := []agent.RunPlanNode{{ID: "coder", Task: coderTask}, {ID: "critic", Task: "review", DependsOn: []string{"coder"}}}
	if len(metadata.Plan) != len(expected) {
		t.Fatal("actual daemon changed task cardinality")
	}
	for i, node := range expected {
		actual, _ := json.Marshal(metadata.Plan[i])
		want, _ := json.Marshal(node)
		if string(actual) != string(want) || metadata.Nodes()[node.ID].State != "succeeded" {
			t.Fatal("actual daemon task or dependency settlement differs")
		}
	}
	records, err := control.ListAuthoritativeRecords("action")
	expectedPhases := 3
	if input.Recovery {
		expectedPhases = 5
	}
	if err != nil || len(records) != expectedPhases {
		t.Fatal("actual daemon durable phases missing", err)
	}
	for _, record := range records {
		lease, active, leaseErr := control.GetActionLease(record.ID)
		resources, resourceErr := control.GetResourceLeases(record.ID)
		if record.State != "SUCCEEDED" || leaseErr != nil || resourceErr != nil || len(resources) != 0 || (active && lease.LeaseUntil > time.Now().Unix()) {
			t.Fatal("actual daemon terminal proof or resource release differs", leaseErr, resourceErr)
		}
	}
	if input.Browser {
		expectedRuntime, expectedSchema := uint64(2), uint32(2)
		if input.Recovery {
			expectedRuntime, expectedSchema = 3, 4
		}
		if metadata.RuntimeVersion() != expectedRuntime || metadata.SchemaVersion != expectedSchema {
			t.Fatal("actual browser confirmation did not retain its runtime revision")
		}
		policies, err := control.ListAuthoritativeRecords("policy")
		if err != nil || len(policies) != 1 {
			t.Fatal("isolated native backup configuration missing", err)
		}
		var policy struct {
			Name       string
			Enabled    bool
			Protection struct {
				Mode       string
				Recipients []string
			}
		}
		if json.Unmarshal(policies[0].Payload, &policy) != nil || policy.Name != "Native browser verification" || !policy.Enabled || policy.Protection.Mode != "age-recipient" ||
			len(policy.Protection.Recipients) != 1 || policy.Protection.Recipients[0] != "age1fu59d59ghmr8x2t5dyzjs9xdcjgnakujp7mjy7cz2v7fq6vjqypskh4e62" {
			t.Fatal("backup policy was not retained by the actual Go daemon")
		}
	}
	result := map[string]any{
		"status": "PASS", "runId": metadata.RunID, "runStatus": metadata.Status, "nextIndex": metadata.NextIndex,
		"durablePhases": len(records), "resourcesReleased": true, "requestSha256": metadata.Request.SHA256,
		"requestLength": metadata.Request.Length, "actualPreset": metadata.Preset, "actualPlan": metadata.Plan,
		"runtimeRevision":                 metadata.RuntimeVersion(),
		"metadataSchemaVersion":           metadata.SchemaVersion,
		"independentlyValidatedByGoStore": true, "independentNativeBackupPolicyVerified": input.Browser,
		"sourceCommit": os.Getenv("DEEPSEEK_TEST_HEAD_SHA")}
	if input.Recovery {
		for key, value := range nativeBrowserRecoveryInspection(t, metadata, records) {
			result[key] = value
		}
	}
	nativeSigningWrite(t, filepath.Join(directory, "daemon-inspected.json"), result)
}

func nativeBrowserRecoveryInspection(t *testing.T, run agent.RunMetadata, records []store.Record) map[string]any {
	t.Helper()
	if run.Recovery == nil || run.Recovery.Generation != 1 || run.Recovery.Metadata == nil ||
		run.Recovery.Metadata.ExpectedIndex != 10 || run.Recovery.Metadata.ExpectedRuntimeRevision != 2 ||
		run.RecoveryProgress == nil || run.RecoveryProgress.Receipt == nil || run.RecoveryProgress.Receipt.Evidence == nil {
		t.Fatal("actual React recovery did not retain typed Go and Rust progress custody")
	}
	proof := run.RecoveryProgress.Receipt
	if err := agentproof.VerifyAgentRecoveryProgress(proof, proof.Evidence.Context, agentproof.Binding{
		PublicKey: os.Getenv("DEEPSEEK_WORKER_AUTHORITY_SIGNER_PUBLIC_KEY"), FleetID: mutationFleetID, Environment: mutationEnvironment,
	}); err != nil {
		t.Fatal("actual React recovery signature was not independently verified", err)
	}
	var task, synthesis string
	prior := 0
	for _, record := range records {
		binding, err := store.AgentRunExecutionBindingFromRecord(record)
		if err != nil {
			t.Fatal(err)
		}
		if binding.Request != run.Request {
			prior++
			continue
		}
		switch binding.Phase {
		case "tasks":
			if task != "" {
				t.Fatal("actual React rerun admitted duplicate task phases")
			}
			task = record.ID
		case "synthesis":
			if synthesis != "" {
				t.Fatal("actual React rerun admitted duplicate synthesis phases")
			}
			synthesis = record.ID
		default:
			t.Fatal("actual React rerun replayed planning or another phase")
		}
	}
	if prior != 3 || task == "" || synthesis == "" || task == synthesis {
		t.Fatal("actual React recovery did not retain exactly three original and two fresh phases")
	}
	raw, err := (proto.MarshalOptions{Deterministic: true}).Marshal(proof)
	if err != nil {
		t.Fatal(err)
	}
	digest := sha256.Sum256(raw)
	return map[string]any{"independentBrowserRecoveryProgressVerified": true, "recoveryGeneration": run.Recovery.Generation,
		"browserRecoveryTaskActionId": task, "browserRecoverySynthesisActionId": synthesis,
		"browserRecoveryReceiptSha256": hex.EncodeToString(digest[:]), "browserRecoveryReceiptBase64": base64.RawURLEncoding.EncodeToString(raw)}
}

// Promote attested empty inventories. No Agent inputs or execution records are
// imported; the isolated backup configuration is created by the Go handler below.
func nativeBrowserEmptyBackupOwnership(t *testing.T, control *store.Control) {
	t.Helper()
	for _, domain := range []string{"policy", "target"} {
		current, err := control.GetCutover(domain)
		if err != nil {
			t.Fatal(err)
		}
		dual, err := control.TransitionCutover(store.CutoverTransition{Domain: domain, To: store.CutoverDualEvaluate,
			ExpectedRevision: current.Revision, ExpectedEpoch: current.Epoch, FencingToken: current.FencingToken,
			TransferID: "browser-empty-" + domain})
		if err != nil {
			t.Fatal(err)
		}
		var imported store.PythonInventoryImportResult
		if domain == "policy" {
			imported = attestedEmptyPolicyImport(t, control)
		} else {
			directory := filepath.Join("..", "store", "testdata", "target-health-empty-v1")
			source := copyPythonSourceFixture(t, filepath.Join("target-health-empty-v1", "python_empty_control_source_v1.sqlite3"), "target")
			scheduler := filepath.Join(filepath.Dir(filepath.Dir(source)), ".backup-scheduler")
			if err := os.MkdirAll(scheduler, 0o700); err != nil {
				t.Fatal(err)
			}
			raw, err := os.ReadFile(filepath.Join(directory, "python_empty_scheduler_source_v1.sqlite3"))
			if err != nil {
				t.Fatal(err)
			}
			if err := os.WriteFile(filepath.Join(scheduler, "scheduler.db"), raw, 0o600); err != nil {
				t.Fatal(err)
			}
			raw, err = os.ReadFile(filepath.Join(directory, "python_empty_target_inventory_export_v2.json"))
			if err != nil {
				t.Fatal(err)
			}
			attestation, err := store.AttestPythonInventorySource(source, raw)
			if err != nil {
				t.Fatal(err)
			}
			imported, err = control.ImportAttestedPythonInventory(raw, attestation)
			if err != nil {
				t.Fatal(err)
			}
		}
		if imported.Imported != 0 {
			t.Fatal("browser ownership import seeded application records")
		}
		request := store.CutoverTransition{Domain: domain, To: store.CutoverGoAuthoritative,
			ExpectedRevision: dual.Revision, ExpectedEpoch: dual.Epoch, FencingToken: dual.FencingToken,
			TransferID: imported.TransferID, Authority: authorityCheckpointFixture(t)}
		if _, err := control.TransitionCutover(signedPromotionRoute(t, request, dual, time.Now().UTC().Unix(), &imported)); err != nil {
			t.Fatal(err)
		}
	}
}

func nativeBrowserConfigureBackupPolicy(t *testing.T, control *store.Control) {
	t.Helper()
	t.Setenv("AUTH_DISABLED", "true")
	public := http.NewServeMux()
	RegisterPublic(public, control)
	body := `{"name":"Native browser verification","enabled":true,"protection":{"mode":"age-recipient","recipients":["age1fu59d59ghmr8x2t5dyzjs9xdcjgnakujp7mjy7cz2v7fq6vjqypskh4e62"]}}`
	request := httptest.NewRequest(http.MethodPost, "http://127.0.0.1/api/workspace/backup-policies", strings.NewReader(body))
	request.Header.Set("Content-Type", "application/json")
	response := httptest.NewRecorder()
	public.ServeHTTP(response, request)
	if response.Code != http.StatusOK {
		t.Fatalf("isolated native backup configuration: %d %s", response.Code, response.Body.String())
	}
	for _, domain := range []string{"agent_run", "action"} {
		records, err := control.ListAuthoritativeRecords(domain)
		if err != nil || len(records) != 0 {
			t.Fatal("backup configuration seeded Agent execution input", domain, err)
		}
	}
}
